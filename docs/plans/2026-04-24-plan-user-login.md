# User Login + Cross-Device Sync + Paid Tiers — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a soft-gate login system with Supabase (Google OAuth + Magic Link OTP), 3-tier subscription (Free / Sync $4 / Pro $12), cross-device sync of user content, and managed AI proxy with per-user quota + rate limiting. BYOK path stays local and untouched throughout.

**Architecture:** Chrome extension calls Supabase (Auth + Postgres + Realtime) directly for sync, Edge Functions for AI proxy + Stripe webhooks, Stripe Checkout + Billing Portal for payments. Config split: `apiKey` stays local, `baseURL`/`model` cloud-synced via `byok_prefs`. Offline writes queue in `chrome.storage.local.sync:queue` and drain on reconnect.

**Tech stack:** TypeScript, React 18, Vite (no bundler change), Supabase JS SDK v2, Deno (Edge Functions), Postgres 15, Stripe Node SDK, Vitest (unit), `supabase start` local stack (integration), `@axe-core/playwright` (a11y CI), `stripe listen` (webhook testing).

**Spec source of truth:** `docs/specs/2026-04-24-spec-user-login.md` — read it before starting any phase. §14 = design-review decisions, §15 = eng-review decisions. Both addenda OVERRIDE conflicts with §1-§13.

```sql
create or replace function increment_ai_usage()
returns int
language plpgsql security definer as $$
declare
  v_uid    uuid := auth.uid();
  v_tier   text;
  v_period text;
  v_limit  int;
  v_used   int;
begin
  if v_uid is null then raise exception 'unauthenticated'; end if;

  select tier into v_tier from subscriptions where user_id = v_uid;
  if v_tier is null then v_tier := 'free'; end if;

  if v_tier = 'sync' then
    raise exception 'sync tier has no managed ai';
  end if;

  v_period := case when v_tier = 'pro'
              then to_char(now() at time zone 'utc', 'YYYY-MM')
              else 'lifetime-trial' end;
  v_limit  := case when v_tier = 'pro' then 300 else 20 end;

  insert into ai_usage (user_id, period, used)
  values (v_uid, v_period, 0)
  on conflict (user_id, period) do nothing;

  select used into v_used from ai_usage
  where user_id = v_uid and period = v_period for update;

  if v_used >= v_limit then return -1; end if;

  update ai_usage set used = used + 1
  where user_id = v_uid and period = v_period;

  return v_used + 1;
end $$;

revoke all on function increment_ai_usage() from public;
grant  execute on function increment_ai_usage() to authenticated;

create or replace function rate_limit_check(
  p_max_count int default 10,
  p_window_sec int default 300
) returns boolean
language plpgsql security definer as $$
declare
  v_uid uuid := auth.uid();
  v_bucket timestamptz;
  v_count int;
begin
  if v_uid is null then raise exception 'unauthenticated'; end if;
  v_bucket := date_trunc('minute', now()) -
              (extract(minute from now())::int % (p_window_sec / 60)) * interval '1 minute';
  insert into rate_limits (user_id, window_start, count)
  values (v_uid, v_bucket, 1)
  on conflict (user_id, window_start) do update
    set count = rate_limits.count + 1
  returning count into v_count;
  if v_count > p_max_count then return false; end if;
  return true;
end $$;

revoke all on function rate_limit_check(int, int) from public;
grant  execute on function rate_limit_check(int, int) to authenticated;
```

```ts
// supabase/functions/_shared/auth.ts
import { createClient, SupabaseClient, User } from 'https://esm.sh/@supabase/supabase-js@2'

const SB_URL = Deno.env.get('SUPABASE_URL')!
const SB_ANON = Deno.env.get('SUPABASE_ANON_KEY')!

export async function getUserFromRequest(req: Request):
  Promise<{ user: User; client: SupabaseClient } | null>
{
  const auth = req.headers.get('Authorization')
  if (!auth) return null
  const client = createClient(SB_URL, SB_ANON, {
    global: { headers: { Authorization: auth } },
  })
  const { data: { user } } = await client.auth.getUser()
  return user ? { user, client } : null
}
```

```ts
// chrome-extension/reader/lib/storage-schema.ts
export type StorageSchema = {
  session:                      unknown        // Supabase session object
  config_apikey:                string         // raw OpenAI key, LOCAL ONLY
  config_prefs:                 { baseURL: string; model: string } | null
  'sync:queue':                 { table: string; op: 'upsert' | 'delete'; row: any; ts: number }[]
  migrationState:               'idle' | 'in-progress' | 'done' | 'paused'
  paperIdMap:                   Record<string, string>  // paper_key → papers.id
  churnModalSeen:               boolean
  libraryCapBannerDismissed:    number | null
  'library:lock':               boolean
}
export type StorageKey = keyof StorageSchema

export async function getItem<K extends StorageKey>(k: K): Promise<StorageSchema[K] | null> {
  const r = await chrome.storage.local.get(k)
  return (r[k] as StorageSchema[K]) ?? null
}

export async function setItem<K extends StorageKey>(k: K, v: StorageSchema[K]): Promise<void> {
  await chrome.storage.local.set({ [k]: v })
}

export async function removeItem<K extends StorageKey>(k: K): Promise<void> {
  await chrome.storage.local.remove(k)
}
```

Note: `paper:{key}:*` keys (parsed/memory/notes/highlights/chat/canvas/summary:v*) remain in the pre-existing `storage.ts` wrapper (§14.10). `storage-schema.ts` governs only the new account-adjacent keys.

```ts
// chrome-extension/reader/lib/sync-queue.ts
import { supabase } from './supabase'
import { getItem, setItem, removeItem } from './storage-schema'

export type PendingOp = {
  table: string
  op: 'upsert' | 'delete'
  row: any
  ts: number
}

export async function enqueue(op: PendingOp): Promise<void> {
  const q = (await getItem('sync:queue')) ?? []
  q.push(op)
  await setItem('sync:queue', q)
}

export async function drain(): Promise<void> {
  if (!navigator.onLine) return
  const q = (await getItem('sync:queue')) ?? []
  for (const op of q) {
    try {
      if (op.op === 'upsert') await supabase.from(op.table).upsert(op.row)
      if (op.op === 'delete') await supabase.from(op.table).delete().eq('id', op.row.id)
    } catch {
      return  // keep queue; retry on next drain
    }
  }
  await removeItem('sync:queue')
}

export async function clear(): Promise<void> {
  await removeItem('sync:queue')
}

if (typeof window !== 'undefined') {
  window.addEventListener('online', () => { void drain() })
}
```

```tsx
export function QuotaChip({ onOpenMenu }: { onOpenMenu: () => void }) {

  const [used, setUsed] = useState(0)
  const [limit, setLimit] = useState(20)

    const refresh = async () => {
      const { data: { session } } = await supabase.auth.getSession()
      const byok = await getItem('config_apikey')

      if (!session) { setState('hidden'); return }

        : 'lifetime-trial'
      const { data: u } = await supabase.from('ai_usage')
        .select('used').eq('period', period).single()
      const _used = u?.used ?? 0
      const _limit = tier === 'pro' ? 300 : 20
      setUsed(_used); setLimit(_limit)

      if (tier === 'pro') { setState('pro'); return }
      const pct = _used / _limit
      if (pct >= 1) setState('free-critical')
      else if (pct >= 0.6) setState('free-warn')
      else setState('free-normal')
    }

    refresh()
    const channel = supabase.channel('quota-chip')
      .on('postgres_changes', {
        event: 'UPDATE', schema: 'public', table: 'ai_usage',
      }, () => void refresh())
      .subscribe()
    return () => { channel.unsubscribe() }

    state === 'sync' ? t('quota.sync') :
    state === 'pro' ? t('quota.pro', { used, limit }) :
    state === 'free-critical' ? t('quota.free.critical') :
    state === 'free-warn' ? t('quota.free.warn', { used, limit }) :
    t('quota.free', { used, limit })
```

```ts
// chrome-extension/reader/lib/migration.ts
import { supabase } from './supabase'
import { getItem, setItem, removeItem } from './storage-schema'

type Progress = { done: number; total: number; phase: 'idle' | 'migrating' | 'done' | 'paused' }
type ProgressListener = (p: Progress) => void

const listeners: ProgressListener[] = []
export function onProgress(fn: ProgressListener) {
  listeners.push(fn)
  return () => { const i = listeners.indexOf(fn); if (i >= 0) listeners.splice(i, 1) }
}
function emit(p: Progress) { listeners.forEach(fn => fn(p)) }

function chunk<T>(arr: T[], size: number): T[][] {
  const out: T[][] = []
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size))
  return out
}

export async function pushLocalToCloud(): Promise<void> {
  await setItem('migrationState', 'in-progress')
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) throw new Error('not authenticated')

  const rawLib = (await chrome.storage.local.get('library'))['library'] ?? []
  const total = rawLib.length
  if (total === 0) {
    await setItem('migrationState', 'done')
    emit({ done: 0, total: 0, phase: 'done' })
    return
  }

  const paperIdMap: Record<string, string> = (await getItem('paperIdMap')) ?? {}
  let done = 0
  emit({ done, total, phase: 'migrating' })

  for (const batch of chunk(rawLib, 50)) {
    const rows = batch.map((p: any) => ({
      user_id: user.id,
      paper_key: p.paperKey ?? p.id,
      title: p.title, authors: p.authors, venue: p.venue,
      pages: p.pages ?? 0, role: p.role, topic: p.topic, judgment: p.judgment,
      added_at: new Date(p.addedAt).toISOString(),
      last_read: new Date(p.lastRead).toISOString(),
    }))
    const { data, error } = await supabase.from('papers')
      .upsert(rows, { onConflict: 'user_id,paper_key' })
      .select('id, paper_key')
    if (error) {
      await setItem('migrationState', 'paused')
      emit({ done, total, phase: 'paused' })
      throw error
    }
    data?.forEach(r => { paperIdMap[r.paper_key] = r.id })
    await setItem('paperIdMap', paperIdMap)
    done += batch.length
    emit({ done, total, phase: 'migrating' })
  }

  for (const [paperKey, paperId] of Object.entries(paperIdMap)) {
    for (const [storageKeyFn, table] of [
      [(k: string) => `paper:${k}:highlights`, 'highlights'],
      [(k: string) => `paper:${k}:notes`,      'margin_notes'],
      [(k: string) => `paper:${k}:memory`,     'memory'],
      [(k: string) => `paper:${k}:chat`,       'chat'],
      [(k: string) => `paper:${k}:canvas`,     'canvas'],
    ] as const) {
      const key = storageKeyFn(paperKey)
      const local = (await chrome.storage.local.get(key))[key]
      if (!local) continue
      const rows = transformForTable(local, paperId, user.id, table)
      if (rows.length === 0) continue
      const conflict = (table === 'memory' || table === 'chat' || table === 'canvas') ? 'paper_id' : 'id'
      await supabase.from(table).upsert(rows as any, { onConflict: conflict })
    }
  }

  await setItem('migrationState', 'done')
  await removeItem('paperIdMap')
  emit({ done, total, phase: 'done' })
}

function transformForTable(local: any, paperId: string, userId: string, table: string): any[] {
  switch (table) {
    case 'highlights':
      return (local as any[]).map(h => ({ ...h, paper_id: paperId, user_id: userId }))
    case 'margin_notes':
      return (local as any[]).map(n => ({
        id: n.id, paper_id: paperId, user_id: userId,
        paragraph_id: n.paragraphId, kind: n.kind, source: n.source, body: n.body,
        created_at: new Date(n.createdAt).toISOString(),
      }))
    case 'memory':
      return [{ paper_id: paperId, user_id: userId, ...local }]
    case 'chat':
      return [{ paper_id: paperId, user_id: userId, turns: local.turns ?? [] }]
    case 'canvas':
      return [{ paper_id: paperId, user_id: userId, nodes: local.nodes ?? [] }]
    default: return []
  }
}

export async function resumeIfNeeded(): Promise<void> {
  const state = await getItem('migrationState')
  if (state === 'in-progress' || state === 'paused') {
    await pushLocalToCloud()
  }
}
```

```ts
export type MigrationChoice = 'merge' | 'local' | 'cloud'

export async function detectAndRoute(): Promise<
  | { route: 'empty' }
  | { route: 'push-local' }
  | { route: 'pull-only' }
  | { route: 'conflict'; stats: { local: number; cloud: number; overlap: number } }
> {
  const { count: cloudCount } = await supabase.from('papers').select('*', { count: 'exact', head: true })
  const rawLib = (await chrome.storage.local.get('library'))['library'] ?? []
  const localCount = rawLib.length

  if ((cloudCount ?? 0) === 0 && localCount === 0) return { route: 'empty' }
  if ((cloudCount ?? 0) === 0 && localCount > 0) return { route: 'push-local' }
  if ((cloudCount ?? 0) > 0 && localCount === 0) return { route: 'pull-only' }

  const { data: cloudPapers } = await supabase.from('papers').select('paper_key')
  const cloudKeys = new Set((cloudPapers ?? []).map((p: any) => p.paper_key))
  const overlap = rawLib.filter((p: any) => cloudKeys.has(p.paperKey ?? p.id)).length
  return { route: 'conflict', stats: { local: localCount, cloud: cloudCount ?? 0, overlap } }
}

export async function applyMergeChoice(choice: MigrationChoice): Promise<void> {
  if (choice === 'merge') { await pushLocalToCloud(); return }
  if (choice === 'local') { await pushLocalToCloud(); /* plus scrub pass — see §7.3 */ return }
  if (choice === 'cloud') {
    // Wipe local paper:* keys; realtime sync will re-populate
    const all = await chrome.storage.local.get(null)
    const toRemove = Object.keys(all).filter(k => k.startsWith('paper:') && !k.endsWith(':parsed'))
    if (toRemove.length) await chrome.storage.local.remove(toRemove)
    await chrome.storage.local.remove('library')
    await setItem('migrationState', 'done')
  }
}
```

```ts
import { json, errorResp } from '../_shared/responses.ts'
import { serviceRoleClient } from '../_shared/clients.ts'

const OPENAI_KEY = Deno.env.get('OPENAI_API_KEY')!
const OPENAI_URL = Deno.env.get('OPENAI_BASE_URL')!
const OPENAI_MODEL = Deno.env.get('OPENAI_MODEL')!
const UPGRADE_URL = Deno.env.get('UPGRADE_URL')!

Deno.serve(async (req) => {
  const session = await getUserFromRequest(req)
  if (!session) return errorResp('Unauthorized', 401)
  const { user, client: supa } = session

  // Rate limit (§15.2)
  const { data: allowed } = await supa.rpc('rate_limit_check', {
    p_max_count: 10, p_window_sec: 300,
  })
  if (allowed === false) {
    return new Response(JSON.stringify({ reason: 'rate-limited' }), {
      status: 429,
      headers: { 'Content-Type': 'application/json', 'Retry-After': '60' },
    })
  }

  // Quota (§8.3)
  let newUsed: number
  try {
    const { data, error } = await supa.rpc('increment_ai_usage')
    if (error) {
      if (error.message.includes('sync tier')) return json({ reason: 'sync-tier-no-managed-ai' }, 403)
      return json({ error: 'quota-check-failed' }, 500)
    }
    newUsed = data as number
  } catch {
    return json({ error: 'quota-check-failed' }, 500)
  }

  if (newUsed === -1) {
    const { data: sub } = await supa.from('subscriptions').select('tier').eq('user_id', user.id).single()
    const tier = sub?.tier ?? 'free'
    const limit = tier === 'pro' ? 300 : 20
    return json({ tier, used: limit, limit, upgrade_url: UPGRADE_URL }, 402)
  }

  const body = await req.json()
  const oaResp = await fetch(`${OPENAI_URL}/chat/completions`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${OPENAI_KEY}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({ model: OPENAI_MODEL, messages: body.messages, stream: true }),
  })

  const admin = serviceRoleClient()
  const { data: sub } = await supa.from('subscriptions').select('tier').eq('user_id', user.id).single()
  EdgeRuntime.waitUntil(
    admin.from('ai_usage_log').insert({
      user_id: user.id, kind: body.kind, tier_at_call: sub?.tier ?? 'free', model: OPENAI_MODEL,
    })
  )

  return new Response(oaResp.body, { headers: { 'Content-Type': 'text/event-stream' } })
})
```

```ts
async function streamBYOK(messages: any[], kind: string, onChunk: (c: string) => void) {
  /* existing logic unchanged — rename the old impl to this */
}

async function streamThroughProxy(messages: any[], kind: string, onChunk: (c: string) => void) {
  const { data: { session } } = await supabase.auth.getSession()
  if (!session) throw new ProxyError('unauthenticated', {})
  const resp = await fetch(`${import.meta.env.VITE_SUPABASE_URL}/functions/v1/ai-proxy`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${session.access_token}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ kind, messages, stream: true }),
  })
  switch (resp.status) {
    case 200: {
      const reader = resp.body!.getReader()
      const decoder = new TextDecoder()
      while (true) {
        const { done, value } = await reader.read()
        if (done) break
        onChunk(decoder.decode(value))
      }
      return
    }
    case 402: throw new ProxyError('quota-exhausted', await resp.json())
    case 403: throw new ProxyError('sync-no-managed-ai', await resp.json())
    case 429: throw new ProxyError('rate-limited', {})
    case 401: throw new ProxyError('unauthenticated', {})
    case 500: throw new ProxyError('server-error', {})
    default:  throw new ProxyError('unknown', {})
  }
}

export class ProxyError extends Error {
  constructor(public code: string, public payload: any) { super(code) }
}

export async function callAI(messages: any[], kind: string, onChunk: (c: string) => void) {
  const apiKey = await getItem('config_apikey')
  if (apiKey) return streamBYOK(messages, kind, onChunk)
  return streamThroughProxy(messages, kind, onChunk)
}
```

**Files:**

- [ ] **Step 1: Full impl** (spec §9.3 + §14.7.7.4 cancel_at_period_end)

```ts
import { serviceRoleClient } from '../_shared/clients.ts'
import Stripe from 'https://esm.sh/stripe@14'

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY')!, { apiVersion: '2023-10-16' })
const WEBHOOK_SECRET = Deno.env.get('STRIPE_WEBHOOK_SECRET')!

Deno.serve(async (req) => {
  const sig = req.headers.get('stripe-signature')
  if (!sig) return new Response('missing sig', { status: 400 })
  const raw = await req.text()
  let event: Stripe.Event
  try {
    event = await stripe.webhooks.constructEventAsync(raw, sig, WEBHOOK_SECRET)
  } catch {
    return new Response('bad sig', { status: 400 })
  }

  const supa = serviceRoleClient()

  switch (event.type) {
    case 'checkout.session.completed': {
      const s = event.data.object as Stripe.Checkout.Session
      const tier = s.metadata?.tier as 'sync' | 'pro'
      const sub = await stripe.subscriptions.retrieve(s.subscription as string)
      await supa.from('subscriptions').upsert({
        user_id: s.client_reference_id,
        tier,
        stripe_customer_id: s.customer,
        stripe_subscription_id: s.subscription,
        current_period_end: new Date(sub.current_period_end * 1000).toISOString(),
        cancel_at_period_end: sub.cancel_at_period_end,
      })
      break
    }
    case 'customer.subscription.updated': {
      const sub = event.data.object as Stripe.Subscription
      await supa.from('subscriptions').update({
        current_period_end: new Date(sub.current_period_end * 1000).toISOString(),
        cancel_at_period_end: sub.cancel_at_period_end,
        canceled_at: sub.cancel_at_period_end ? new Date().toISOString() : null,
      }).eq('stripe_subscription_id', sub.id)
      break
    }
    case 'customer.subscription.deleted': {
      const sub = event.data.object as Stripe.Subscription
      await supa.from('subscriptions').update({
        tier: 'free',
        stripe_subscription_id: null,
        current_period_end: null,
        cancel_at_period_end: false,
      }).eq('stripe_subscription_id', sub.id)
      break
    }
  }

  return new Response('ok', { status: 200 })
})
```

```tsx
// chrome-extension/reader/components/churn-modal.tsx
import React, { useEffect, useState } from 'react'
import { supabase } from '../lib/supabase'
import { getItem, setItem } from '../lib/storage-schema'
import { t } from '../lib/i18n'

export function ChurnModal() {
  const [visible, setVisible] = useState(false)

  useEffect(() => {
    void (async () => {
      const seen = await getItem('churnModalSeen')
      if (seen) return
      const { data: sub } = await supabase.from('subscriptions').select('tier, canceled_at').single()
      if (sub?.tier !== 'free') return
      if (!sub.canceled_at) return
      const daysSince = (Date.now() - new Date(sub.canceled_at).getTime()) / 86400000
      if (daysSince > 7) return
      setVisible(true)
    })()
  }, [])

  async function close() {
    await setItem('churnModalSeen', true)
    setVisible(false)
  }

  async function restore() {
    await close()
    const { data } = await supabase.functions.invoke('create-portal-session')
    if (data?.url) chrome.tabs.create({ url: data.url })
  }

  if (!visible) return null
  return (
    <div role="dialog" aria-modal="true" aria-labelledby="churn-title"
         className="modal-backdrop" onClick={close}>
      <div className="modal-panel" onClick={e => e.stopPropagation()}>
        <h2 id="churn-title">{t('churn.headline')}</h2>
        <p>{t('churn.body')}</p>
        <div className="actions">
          <button className="primary" onClick={restore}>{t('churn.restore')}</button>
          <button onClick={close}>{t('churn.later')}</button>
        </div>
      </div>
    </div>
  )
}
```

```ts
// chrome-extension/reader/lib/focus-trap.ts
export function trapFocus(el: HTMLElement): () => void {
  const focusable = el.querySelectorAll<HTMLElement>('button, input, [tabindex]:not([tabindex="-1"]), a')
  if (focusable.length === 0) return () => {}
  const first = focusable[0]
  const last = focusable[focusable.length - 1]
  const handler = (e: KeyboardEvent) => {
    if (e.key !== 'Tab') return
    if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus() }
    else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus() }
  }
  el.addEventListener('keydown', handler)
  return () => el.removeEventListener('keydown', handler)
}
```
